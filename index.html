router.post('/webhook/alert', async (req, res) => {
  const incident = await validateTelemetry(req.body);
  if (incident.gasSpike || incident.flameSpike) {
    await sqlite.incidents.insert({
      payload: incident,
      retry: 'exponential-backoff'
    });
  }
  res.status(202).json({ accepted: true });
});
